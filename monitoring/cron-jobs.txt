# AI Emulators Ecosystem - Cron Jobs for Automated Maintenance
# Install with: crontab -e (add the following lines)

# Backup ecosystem daily at 2 AM
0 2 * * * /opt/ai-ecosystem/future-app/scripts/backup-ecosystem.sh backup

# Log rotation daily at 3 AM
0 3 * * * /opt/ai-ecosystem/future-app/scripts/manage-logs.sh rotate

# Log cleanup weekly on Sundays at 4 AM
0 4 * * 0 /opt/ai-ecosystem/future-app/scripts/manage-logs.sh cleanup

# System health check every 15 minutes
*/15 * * * * /opt/ai-ecosystem/future-app/scripts/check-status.sh > /opt/ai-ecosystem/logs/health-check-$(date +\%Y\%m\%d).log 2>&1

# Docker cleanup weekly on Sundays at 5 AM
0 5 * * 0 docker system prune -f --volumes

# Update monitoring dashboards weekly
0 6 * * 0 curl -X POST http://localhost:3020/api/admin/provisioning/dashboards/reload

# Backup verification daily at 6 AM
0 6 * * * /opt/ai-ecosystem/future-app/scripts/backup-ecosystem.sh verify
